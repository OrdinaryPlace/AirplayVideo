cmake_minimum_required(VERSION 3.20)
project(AirplayVideo VERSION 0.2.11 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavdevice libavformat libavcodec libavfilter libavutil libswscale libswresample libva)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
pkg_check_modules(PANGO REQUIRED IMPORTED_TARGET pangocairo)
add_library(pair_ap STATIC vendor/pair_ap/pair.c vendor/pair_ap/pair-tlv.c vendor/pair_ap/pair_fruit.c vendor/pair_ap/pair_homekit.c)
target_include_directories(pair_ap PUBLIC vendor/pair_ap)
target_compile_definitions(pair_ap PRIVATE CONFIG_OPENSSL=1)
target_link_libraries(pair_ap PUBLIC OpenSSL::Crypto PkgConfig::PLIST PkgConfig::SODIUM)
# DEBUG_PAIR must never be enabled: it logs authentication material.
add_library(airplay_core src/common.cpp src/net.cpp src/discovery.cpp src/pairing.cpp src/video.cpp src/generated.cpp src/audio.cpp src/mirror.cpp src/media.cpp src/recording.cpp src/stream.cpp)
target_include_directories(airplay_core PUBLIC include)
target_compile_options(airplay_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(airplay_core PUBLIC pair_ap PkgConfig::AV PkgConfig::PANGO Threads::Threads)
add_executable(airplayvideo-engine src/main.cpp)
target_link_libraries(airplayvideo-engine PRIVATE airplay_core PkgConfig::HTTP)
enable_testing()
add_executable(protocol-tests tests/protocol.cpp)
target_link_libraries(protocol-tests PRIVATE airplay_core)
add_test(NAME protocol COMMAND protocol-tests)
add_executable(media-tests tests/media.cpp)
add_executable(measure-sync tests/measure-sync.cpp)
target_link_libraries(measure-sync PRIVATE PkgConfig::AV)
target_link_libraries(media-tests PRIVATE airplay_core)
add_test(NAME media COMMAND media-tests)
add_executable(frame-rate-tests tests/frame-rate.cpp)
target_include_directories(frame-rate-tests PRIVATE include)
target_link_libraries(frame-rate-tests PRIVATE PkgConfig::AV)
add_test(NAME frame-rate COMMAND frame-rate-tests)
add_executable(video-rate-tests tests/video-rate.cpp)
target_link_libraries(video-rate-tests PRIVATE airplay_core)
add_test(NAME video-rate COMMAND video-rate-tests)
add_executable(audio-tests tests/audio.cpp)
target_link_libraries(audio-tests PRIVATE airplay_core)
add_test(NAME audio COMMAND audio-tests)
add_executable(sync-content-tests tests/sync-content.cpp)
target_link_libraries(sync-content-tests PRIVATE airplay_core)
add_test(NAME sync-content COMMAND sync-content-tests)
set_tests_properties(sync-content PROPERTIES TIMEOUT 60)
add_test(NAME file-source COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/file-source.sh $<TARGET_FILE:media-tests>)
add_executable(broadcast-tests tests/broadcast.cpp)
target_link_libraries(broadcast-tests PRIVATE airplay_core)
add_test(NAME broadcast-source COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/broadcast-source.sh $<TARGET_FILE:broadcast-tests>)
set_tests_properties(broadcast-source PROPERTIES TIMEOUT 30)
add_executable(generated-tests tests/generated.cpp)
target_link_libraries(generated-tests PRIVATE airplay_core)
add_test(NAME generated COMMAND generated-tests)
add_executable(recording-tests tests/recording.cpp)
target_link_libraries(recording-tests PRIVATE airplay_core)
add_test(NAME recording COMMAND recording-tests)
set_tests_properties(recording PROPERTIES TIMEOUT 20)
set_tests_properties(generated PROPERTIES TIMEOUT 20)
set_tests_properties(media PROPERTIES TIMEOUT 20)
set_tests_properties(file-source PROPERTIES TIMEOUT 40)
install(TARGETS airplayvideo-engine RUNTIME DESTINATION bin)
